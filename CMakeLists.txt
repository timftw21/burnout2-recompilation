    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/Zc:__cplusplus;/utf-8>")
        if(B2_AVX2 AND NOT b2_portable_default)
            target_compile_options(${target} PRIVATE "$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:CXX>>:/arch:AVX2>")
        endif()
        if(target STREQUAL "b2")
            if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
                target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/MP2>")
            endif()
            target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/DEBUG:FULL;/OPT:REF;/OPT:ICF>")
        endif()
    endif()
    if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_LINKER_TYPE STREQUAL "LLD" AND
       target MATCHES "^b2(-tool)?$")
        target_link_options(${target} PRIVATE
            "$<$<CONFIG:Release>:/opt:lldlto=3>"
            "$<$<CONFIG:Release>:/lldltocache:lto-cache/${target}/Release>"
            "$<$<CONFIG:Release>:/lldltocachepolicy:cache_size_bytes=3g>")
    endif()
    if(NOT WIN32)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fms-extensions;-Wno-ignored-attributes>")
        if(B2_AVX2 AND NOT b2_portable_default)
            target_compile_options(${target} PRIVATE "$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:CXX>>:-mavx2>")
        endif()
        if(NOT APPLE AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_LINKER_TYPE STREQUAL "LLD" AND target MATCHES "^b2(-tool)?$")
            target_link_options(${target} PRIVATE
                "$<$<CONFIG:Release>:LINKER:--thinlto-jobs=6>"
                "$<$<CONFIG:Release>:LINKER:--thinlto-cache-dir=lto-cache/${target}/Release>"
                "$<$<CONFIG:Release>:LINKER:--thinlto-cache-policy=cache_size_bytes=3g>")
        endif()
    endif()
endforeach()

            message(FATAL_ERROR "Clang's x64 profile runtime is missing.")
        endif()
        foreach(target IN ITEMS b2 b2-tool)
            if(TARGET ${target})
                target_link_directories(${target} PRIVATE "${b2_clang_resources}/lib/windows")
            endif()
        endforeach()
        else()
            set(b2_profile_option "-fprofile-generate=${CMAKE_CURRENT_BINARY_DIR}/pgo")
            foreach(target IN ITEMS b2 b2-tool)
                if(TARGET ${target})
                    target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:${b2_profile_option}>")
                endif()
            endforeach()
        endif()
    else()
        if(WIN32)
            set(b2_profile_option "/clang:-fprofile-use=${B2_PROFILE_USE}")
        else()
            set(b2_profile_option "-fprofile-use=${B2_PROFILE_USE}")
        endif()
    endif()
    foreach(target IN ITEMS b2-core b2-portable-cpu b2-cpu b2-input b2-graphics b2-app b2-tool b2)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE "$<$<AND:$<CONFIG:Release>,$<COMPILE_LANGUAGE:CXX>>:${b2_profile_option}>")
        endif()
    endforeach()
endif()

        if(TARGET ${target})
            set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
        endif()
    endforeach()
endif()
